Complete Controller Template
@RestController @RequestMapping("/api/v1/customers") public class CustomerController { private static final AtomicInteger counter = new AtomicInteger(1003); private final List<Customer> customers = new ArrayList<>(List.of(...)); @GetMapping public List<Customer> getAllCustomers() { return customers; } @GetMapping("/{customerId}") public ResponseEntity<Customer> getCustomer(@PathVariable String customerId) { return customers.stream().filter(c -> c.getCustomerId().equals(customerId)) .findFirst().map(ResponseEntity::ok).orElse(ResponseEntity.notFound().build()); } @PostMapping public ResponseEntity<Customer> createCustomer(@RequestBody Customer customer) { customer.setCustomerId("CUST" + counter.incrementAndGet()); customers.add(customer); return ResponseEntity.status(201).body(customer); } @PutMapping("/{customerId}") public ResponseEntity<Customer> replaceCustomer(@PathVariable String customerId, @RequestBody Customer updated) { for (int i = 0; i < customers.size(); i++) { if (customers.get(i).getCustomerId().equals(customerId)) { updated.setCustomerId(customerId); customers.set(i, updated); return ResponseEntity.ok(updated); } } return ResponseEntity.notFound().build(); } @PatchMapping("/{customerId}/email") public ResponseEntity<Customer> updateEmail(@PathVariable String customerId, @RequestBody Customer patch) { for (Customer c : customers) { if (c.getCustomerId().equals(customerId)) { c.setEmail(patch.getEmail()); return ResponseEntity.ok(c); } } return ResponseEntity.notFound().build(); } @DeleteMapping("/{customerId}") public ResponseEntity<Void> deleteCustomer(@PathVariable String customerId) { boolean removed = customers.removeIf(c -> c.getCustomerId().equals(customerId)); return removed ? ResponseEntity.noContent().build() : ResponseEntity.notFound().build(); } }
📌 This is the complete Day 2 CustomerController. Each endpoint maps to exactly one HTTP method + URL pattern. You'll extend this pattern to AccountController in Exercise 4.
